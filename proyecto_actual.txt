object Fechas {


    fun haceCuanto(tiempo: Long): String{
        val minutos =(System.currentTimeMillis()-tiempo)/60_000
        return when{
            minutos <1-> "hace un momento"
            minutos < 60 -> "hace $minutos min"
            minutos < 1440 -> "hace ${minutos / 60} h"
            else -> "hace ${minutos / 1440} días"
        }

    }
}

package com.reeloop.app.navigation

import androidx.compose.foundation.layout.padding
import androidx.compose.material3.Scaffold
import androidx.compose.runtime.Composable
import androidx.compose.runtime.collectAsState
import androidx.compose.runtime.getValue
import androidx.compose.ui.Modifier
import androidx.lifecycle.viewmodel.compose.viewModel
import androidx.navigation.NavType
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.currentBackStackEntryAsState
import androidx.navigation.compose.rememberNavController
import androidx.navigation.navArgument
import com.reeloop.app.presentation.common.BarraInferior
import com.reeloop.app.presentation.event.ObjetoEvent
import com.reeloop.app.presentation.screens.DetalleScreen
import com.reeloop.app.presentation.screens.ExplorarScreen
import com.reeloop.app.presentation.screens.ImpactoScreen
import com.reeloop.app.presentation.screens.InicioScreen
import com.reeloop.app.presentation.screens.LoginScreen
import com.reeloop.app.presentation.screens.MisObjetosScreen
import com.reeloop.app.presentation.screens.PasaporteScreen
import com.reeloop.app.presentation.screens.PerfilScreen
import com.reeloop.app.presentation.screens.PublicarScreen
import com.reeloop.app.presentation.screens.CodigoQrScreen
import com.reeloop.app.presentation.screens.EscanearQrScreen
import com.reeloop.app.presentation.screens.RegistroScreen
import com.reeloop.app.presentation.screens.SolicitudesScreen
import com.reeloop.app.presentation.viewmodel.ObjetoViewModel
import com.reeloop.app.ui.theme.Fondo

@Composable
fun AppNavigation() {
    val nav = rememberNavController()
    val vm: ObjetoViewModel = viewModel()
    val state by vm.state.collectAsState()
    val entrada by nav.currentBackStackEntryAsState()
    val ruta = entrada?.destination?.route

    val irAPestana: (String) -> Unit = { destino ->
        nav.navigate(destino) {
            popUpTo(Rutas.INICIO) { saveState = true }
            launchSingleTop = true
            restoreState = true
        }
    }

    Scaffold(
        containerColor = Fondo,
        bottomBar = {
            if (ruta != null && ruta !in Rutas.sinBarra) {
                BarraInferior(seleccionada = Rutas.pestanaDe(ruta), onClick = irAPestana)
            }
        }
    ) { padding ->
        NavHost(
            navController = nav,
            startDestination = Rutas.LOGIN,
            modifier = Modifier.padding(padding)
        ) {
            composable(Rutas.LOGIN) {
                LoginScreen(
                    onEntrar = {
                        nav.navigate(Rutas.INICIO) {
                            popUpTo(Rutas.LOGIN) { inclusive = true }
                        }
                    },
                    onCrearCuenta = { nav.navigate(Rutas.REGISTRO) }
                )
            }
            composable(Rutas.REGISTRO) {
                RegistroScreen(
                    onRegistrarme = {
                        nav.navigate(Rutas.INICIO) {
                            popUpTo(Rutas.LOGIN) { inclusive = true }
                        }
                    },
                    onYatengoCuenta = { nav.popBackStack() }
                )
            }
            composable(Rutas.INICIO) {
                InicioScreen(
                    state = state,
                    onPublicar = { nav.navigate(Rutas.PUBLICAR) },
                    onExplorar = { nav.navigate(Rutas.EXPLORAR) },
                    onMisObjetos = { nav.navigate(Rutas.MIS_OBJETOS) },
                    onSolicitudes = { nav.navigate(Rutas.SOLICITUDES) },
                    onEscanear = { nav.navigate(Rutas.ESCANEAR_QR) }
                )
            }
            composable(Rutas.EXPLORAR) {
                ExplorarScreen(
                    state = state,
                    onEvent = vm::onEvent,
                    onVerDetalle = { id -> nav.navigate(Rutas.detalle(id)) }
                )
            }
            composable(
                route = Rutas.DETALLE,
                arguments = listOf(navArgument("id") { type = NavType.IntType })
            ) { backStack ->
                val id = backStack.arguments?.getInt("id") ?: 0
                DetalleScreen(
                    objeto = state.listaObjeto.find { it.id == id },
                    onVolver = { nav.popBackStack() },
                    onSolicitar = { vm.onEvent(ObjetoEvent.OnSolicitar(id)) },
                    onPasaporte = { nav.navigate(Rutas.pasaporte(id)) }
                )
            }
            composable(Rutas.PUBLICAR) {
                PublicarScreen(
                    state = state,
                    onEvent = vm::onEvent,
                    onPublicado = { 
                        nav.navigate(Rutas.MIS_OBJETOS) {
                            popUpTo(Rutas.INICIO)
                        } 
                    }
                )
            }
            composable(Rutas.MIS_OBJETOS) {
                MisObjetosScreen(
                    state = state,
                    onEvent = vm::onEvent,
                    onPublicar = { nav.navigate(Rutas.PUBLICAR) },
                    onVerDetalle = { id -> nav.navigate(Rutas.detalle(id)) }
                )
            }
            composable(
                route = Rutas.CODIGO_QR,
                arguments = listOf(navArgument("id") { type = NavType.IntType })
            ) { backStack ->
                val id = backStack.arguments?.getInt("id") ?: 0
                CodigoQrScreen(
                    objeto = state.listaObjeto.find { it.id == id },
                    onVolver = { nav.popBackStack() }
                )
            }
            composable(Rutas.ESCANEAR_QR) {
                EscanearQrScreen(onVolver = { nav.popBackStack() })
            }
            composable(
                route = Rutas.PASAPORTE,
                arguments = listOf(navArgument("id") { type = NavType.IntType })
            ) { backStack ->
                val id = backStack.arguments?.getInt("id") ?: 0
                PasaporteScreen(
                    objeto = state.listaObjeto.find { it.id == id },
                    onVolver = { nav.popBackStack() }
                )
            }
            composable(Rutas.IMPACTO) {
                ImpactoScreen(state = state)
            }
            composable(Rutas.PERFIL) {
                PerfilScreen(
                    state = state,
                    onMisObjetos = { nav.navigate(Rutas.MIS_OBJETOS) },
                    onSolicitudes = { nav.navigate(Rutas.SOLICITUDES) },
                    onCerrarSesion = {
                        nav.navigate(Rutas.LOGIN) {
                            popUpTo(Rutas.INICIO) { inclusive = true }
                            launchSingleTop = true
                        }
                    }
                )
            }
            composable(Rutas.SOLICITUDES) {
                SolicitudesScreen(
                    state = state,
                    onEvent = vm::onEvent,
                    onVolver = { nav.popBackStack() },
                    onVerQr = { id -> nav.navigate(Rutas.codigoQr(id)) }
                )
            }
        }
    }
}

package com.reeloop.app.navigation

//cada valor es una ruta de la app l, que redirigue a una pantalla

object Rutas {
    const val LOGIN= "login"
    const val REGISTRO ="registro"

    const val MIS_OBJETOS = "mis_objetos"

    const val CODIGO_QR = "codigo_qr/{id}"

import kotlinx.coroutines.flow.StateFlow

import kotlinx.coroutines.flow.stateIn

//Simulaciom
private val solicitantes = listOf("Carlos Mendoza", "Lucía Herrera", "Diego Ramos")
class ObjetoViewModel(application: Application) : AndroidViewModel(application){
    private val repository: ObjetoRepository =
        ObjetoRepositoryImpl(AppDatabase.getDatabase(application).objetoDao())
    private val _state = MutableStateFlow(ObjetoState())

    val state: StateFlow<ObjetoState> = combine(_state, repository.getObjetos()) { estado, lista ->
        estado.copy(listaObjeto = lista)
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), ObjetoState())

    init {
        cargarDatosDeMuestra()
    }

    fun onEvent(event: ObjetoEvent) {
        when (event) {
            is ObjetoEvent.OnExitoVisto -> _state.update { it.copy(exito = false) }
            is ObjetoEvent.OnNombreChange -> _state.update { it.copy(nombre = event.valor, error = "") }
            is ObjetoEvent.OnDescripcionChange -> _state.update { it.copy(descripcion = event.valor, error = "") }
            is ObjetoEvent.OnCategoriaChange -> _state.update { it.copy(categoria = event.valor) }
            is ObjetoEvent.OnConservacionChange -> _state.update { it.copy(conservacion = event.valor) }
            is ObjetoEvent.OnPesoChange -> _state.update { it.copy(peso = event.valor, error = "") }
            is ObjetoEvent.OnModalidadChange -> _state.update { it.copy(modalidad = event.valor) }
            is ObjetoEvent.OnBusquedaChange -> _state.update { it.copy(busqueda = event.valor) }
            is ObjetoEvent.OnFiltroModalidad -> _state.update { it.copy(filtroModalidad = event.valor) }
            is ObjetoEvent.OnSolicitar -> solicitar(event.id)
            is ObjetoEvent.OnAceptar -> cambiarEstado(event.id) { it.copy(estado = Opciones.ACEPTADO) }
            is ObjetoEvent.OnRechazar -> cambiarEstado(event.id) {
                it.copy(estado = Opciones.DISPONIBLE, solicitante = "", fechaSolicitado = 0L)
            }
            ObjetoEvent.OnPublicar -> publicar()
            ObjetoEvent.OnPublicadoConsumido -> _state.update { it.copy(publicado = false) }
        }
    }

    private fun publicar() {
        val s = _state.value
        val peso = s.peso.replace(',', '.').toDoubleOrNull()

        val error = when {
            s.nombre.isBlank() -> "Escribe el nombre del objeto"
            s.descripcion.isBlank() -> "Escribe una descripción"
            peso == null || peso <= 0 -> "Ingresa un peso válido"
            else -> ""
        }
        if (error.isNotEmpty()) {
            _state.update { it.copy(error = error) }
            return
        }


        viewModelScope.launch {
            repository.insertar(
                Objeto(
                    nombre = s.nombre.trim(),
                    descripcion = s.descripcion.trim(),
                    categoria = s.categoria,
                    conservacion = s.conservacion,
                    peso = peso ?: 0.0,
                    modalidad = s.modalidad
                )
            )
            _state.update { ObjetoState().copy(publicado = true, exito = true) }        }
    }
    // Busca el objeto, le aplica el cambio y lo guarda
    private fun cambiarEstado(id: Int, cambio: (Objeto) -> Objeto) {

        viewModelScope.launch { repository.actualizar(cambio(objeto)) }
    }
    private fun solicitar(id: Int) {
        val objeto = state.value.listaObjeto.find { it.id == id } ?: return

                    solicitante = solicitantes[id % solicitantes.size],

                )
            )
        }
    }

    private fun cargarDatosDeMuestra() {

// Cada categoría siempre recibe el mismo color de la paleta
fun colorCategoria(categoria: String): Color {
    val paleta = listOf(Crema, VerdeClaro, AzulClaro, Durazno)
    return paleta[(categoria.hashCode() and Int.MAX_VALUE) % paleta.size]
}

@Composable
fun ObjetoItem(
    objeto: Objeto,
    modifier: Modifier = Modifier,
    mostrarFecha: Boolean = false,
    onClick: () -> Unit = {}
) {
    val forma = RoundedCornerShape(20.dp)

    Row(
        modifier = modifier
            .fillMaxWidth()
            .clip(forma)
            .background(Color.White)
            .border(1.dp, Borde, forma)
            .clickable(onClick = onClick)
            .padding(12.dp),
        horizontalArrangement = Arrangement.spacedBy(14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Box(
            contentAlignment = Alignment.Center,
            modifier = Modifier
                .size(72.dp)
                .background(colorCategoria(objeto.categoria), RoundedCornerShape(16.dp))
        ) {
            Text(
                text = objeto.nombre.take(1).uppercase(),
                fontSize = 28.sp,
                fontWeight = FontWeight.Bold,
                color = TextoOscuro
            )
        }
        Column(verticalArrangement = Arrangement.spacedBy(4.dp)) {
            Text(objeto.nombre, fontSize = 18.sp, fontWeight = FontWeight.Bold, color = TextoOscuro)
            Text("${objeto.categoria} · ${objeto.conservacion}", fontSize = 14.sp, color = TextoGris)
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                Etiqueta(objeto.modalidad, destacada = true)
                Etiqueta(objeto.estado)
            }
            if (mostrarFecha) {
                Text("Publicada ${Fechas.haceCuanto(objeto.fechaPublicado)}", fontSize = 12.sp, color = TextoGris)
            }
        }
    }
}
